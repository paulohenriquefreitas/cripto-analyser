import { Alert, Button, Checkbox, FormControlLabel, MenuItem, Stack, TextField, Typography } from '@mui/material';
import { useMemo, useState } from 'react';
import type { ChartSink } from '@/features/win/models/mt5Intrabar';
import type { CandleView } from '../api/threeCandlesApi';
import { analyzeFibonacci, fibVersion, observeFibonacci, type FibAnchor, type FibVersion } from '../models/FibonacciMovementEngine';
import { ThreeCandlesChart } from './ThreeCandlesChart';
import { type FibonacciOverlay } from './FibonacciPrimitive';
import { DowDebugPanel } from './DowDebugPanel';

const price = (n: number) => n.toLocaleString('pt-BR', { maximumFractionDigits: 2 });
const reasons: Record<string,string> = {
  CONFIRMED_STRUCTURAL_REVERSAL_UP: 'Novo movimento de alta por topo superior confirmado',
  CONFIRMED_STRUCTURAL_REVERSAL_DOWN: 'Novo movimento de baixa por fundo inferior confirmado',
  INITIAL_OPPOSITE_PAIR_HYPOTHESIS: 'Primeira dupla oposta confirmada (hipótese)',
  CONFIRMED_HIGHER_TOP_HYPOTHESIS: 'Ampliação por topo mais alto confirmado (hipótese)',
  CONFIRMED_LOWER_BOTTOM_HYPOTHESIS: 'Ampliação por fundo mais baixo confirmado (hipótese)',
  USER_SELECTED_ANCHORS: 'Âncoras escolhidas pelo usuário',
};
export function ThreeCandlesWorkspace({ controller, view, candleDisplayTimeZone }: {
  controller: { attachChart: (sink: ChartSink) => () => void }; view?: CandleView; candleDisplayTimeZone: string;
  candlesOnly?: boolean; structuralLevels?: boolean; visibleCandleViewport?: boolean;
}) {
  const [visible,setVisible] = useState(true), [compare,setCompare] = useState(false);
  const [mode,setMode] = useState<'MANUAL'|'EXPERIMENTAL'>('EXPERIMENTAL');
  const [scope,setScope] = useState<5|60>(5), [tolerance,setTolerance] = useState(5);
  const [highId,setHighId] = useState(''), [lowId,setLowId] = useState('');
  const [pick,setPick] = useState<'TOP'|'BOTTOM'>(), [manual,setManual] = useState<FibVersion[]>([]);
  const [error,setError] = useState('');
  const [showDow,setShowDow]=useState(false),[dowLabels,setDowLabels]=useState(true),[selectedDow,setSelectedDow]=useState<string>();
  const [showMicro,setShowMicro]=useState(false),[microLabels,setMicroLabels]=useState(false);
  const [allMicroFractals,setAllMicroFractals]=useState(false);
  const microDowOverlay=useMemo(()=>({snapshot:view?.microDow,visible:showMicro,labels:microLabels,allFractals:allMicroFractals}),[view?.microDow,showMicro,microLabels,allMicroFractals]);
  const dowOverlay=useMemo(()=>({snapshot:view?.dow,visible:showDow,labels:dowLabels}),[view?.dow,showDow,dowLabels]);
  const [showDowFifty,setShowDowFifty]=useState(false),[selectedFifty,setSelectedFifty]=useState<string>();
  const dowFiftyOverlay=useMemo(()=>({snapshot:view?.dowFifty,visible:showDowFifty}),[view?.dowFifty,showDowFifty]);
  const fifty=view?.dowFifty;
  const currentFifty=fifty?.references.find(r=>r.wave.id===fifty.currentWaveId);
  const inspectedFifty=fifty?.references.find(r=>r.wave.id===(selectedFifty??fifty.currentWaveId))??currentFifty;
  const fiftyState:Record<string,string>={AGUARDANDO_ONDA:'Aguardando onda',ONDA_CONFIRMADA:'Onda confirmada',MONITORANDO_50:'Monitorando 50%',ROMPIMENTO_EM_AVALIACAO:'Rompimento em avaliacao',REJEICAO_EM_AVALIACAO:'Rejeicao em avaliacao',ONDA_ENCERRADA:'Onda encerrada'};
  const analysis = useMemo(()=>analyzeFibonacci(view,tolerance),[view,tolerance]);
  const lane = analysis.lanes.find(l=>l.timeframeMinutes===scope)!;
  const automatic = lane.versions.at(-1);
  const manualKnown = manual.filter(v=>v.decidedAtMsc<=analysis.cutoff);
  const manualCurrent = manualKnown.filter(v=>v.timeframeMinutes===scope).slice().sort((a,b)=>a.decidedAtMsc-b.decidedAtMsc).at(-1);
  const anchors = analysis.anchors.filter(a=>a.timeframeMinutes===scope);
  const primary = mode==='MANUAL'?manualCurrent:automatic;
  const comparison = compare ? mode==='MANUAL'?automatic:manualCurrent : undefined;
  const overlay: FibonacciOverlay = useMemo(()=>({ primary:visible?primary:undefined, comparison:visible?comparison:undefined, anchors, pick }),[primary,comparison,anchors,pick,visible]);
  const time = (t:number) => new Intl.DateTimeFormat('pt-BR',{timeZone:candleDisplayTimeZone,dateStyle:'short',timeStyle:'short'}).format(new Date(t));
  const selectAnchor = (a:FibAnchor) => { if (pick==='TOP')setHighId(a.id); else if(pick==='BOTTOM')setLowId(a.id); setPick(undefined);setError(''); };
  const applyManual = () => {
    try {
      const high=anchors.find(a=>a.id===highId), low=anchors.find(a=>a.id===lowId);
      if(!high||!low)throw new Error('Escolha um topo e um fundo já confirmados da mesma amplitude.');
      const version=fibVersion(high,low,analysis.cutoff,'MANUAL','USER_SELECTED_ANCHORS',manualCurrent,manual.length+1);
      setManual(previous=>[...previous,version]);setMode('MANUAL');setError('');setPick(undefined);
    }catch(e){setError(String(e instanceof Error?e.message:e));}
  };
  const audit=[...lane.versions,...manualKnown.filter(v=>v.timeframeMinutes===scope)].sort((a,b)=>a.decidedAtMsc-b.decidedAtMsc);
  const manualObservations=observeFibonacci(manualKnown.filter(v=>v.timeframeMinutes===scope).slice().sort((a,b)=>a.decidedAtMsc-b.decidedAtMsc),
    view?.candles.filter(c=>c.availableAtMsc<=analysis.cutoff)??[],tolerance);
  const observations=[...lane.observations,...manualObservations].sort((a,b)=>a.atMsc-b.atMsc);
  return <Stack spacing={1}>
    <Stack direction="row" gap={2} flexWrap="wrap" alignItems="center">
      <FormControlLabel label="Fibonacci" control={<Checkbox checked={visible} onChange={(_,v)=>setVisible(v)} />} />
      <TextField select label="Modo Fibonacci" value={mode} onChange={e=>setMode(e.target.value as typeof mode)} sx={{minWidth:230}}>
        <MenuItem value="EXPERIMENTAL">Automático experimental</MenuItem><MenuItem value="MANUAL">Manual</MenuItem>
      </TextField>
      <TextField select label="Amplitude Fibonacci" value={scope} onChange={e=>{setScope(Number(e.target.value) as 5|60);setHighId('');setLowId('');setPick(undefined);}} sx={{minWidth:190}}>
        <MenuItem value={5}>Micro M5</MenuItem><MenuItem value={60}>Macro H1 2+2</MenuItem>
      </TextField>
      <FormControlLabel label="Comparar manual e automático" control={<Checkbox checked={compare} onChange={(_,v)=>setCompare(v)} />} />
      <TextField select label="Aproximação 50% (pontos)" value={tolerance} onChange={e=>setTolerance(Number(e.target.value))} sx={{minWidth:220}}>
        {[0,5,15,30].map(n=><MenuItem key={n} value={n}>{n}</MenuItem>)}
      </TextField>
    </Stack>
    <Typography variant="caption">Estudo experimental, sem sinal operacional. Extremos confirmados ampliam o movimento; um extremo contrário que supera a amplitude inicia outro movimento a partir do extremo terminal anterior.
      {' '}Toques em 50% não trocam âncoras. Extremos internos ou iguais não definem reversão.</Typography>
    {!analysis.fullMemory && view?.structure && <Alert severity="info">Memória completa indisponível nesta resposta: usando os pivôs M5 e candidatos H1 originais disponíveis. H1 intradiário requer o backend atualizado.</Alert>}
    <Stack direction="row" gap={1} flexWrap="wrap" alignItems="center">
      <Button onClick={()=>setPick('TOP')} disabled={!anchors.some(a=>a.type==='TOP')}>Selecionar máxima no gráfico</Button>
      <Button onClick={()=>setPick('BOTTOM')} disabled={!anchors.some(a=>a.type==='BOTTOM')}>Selecionar mínima no gráfico</Button>
      <Button onClick={()=>setPick(undefined)} disabled={!pick}>Cancelar seleção</Button>
      <TextField select label="Máxima manual confirmada" value={anchors.some(a=>a.id===highId)?highId:''} onChange={e=>setHighId(e.target.value)} sx={{minWidth:235}}>
        <MenuItem value="">Selecione um topo</MenuItem>{anchors.filter(a=>a.type==='TOP').map(a=><MenuItem key={a.id} value={a.id}>{price(a.price)} · {time(a.extremeTimeMsc)}</MenuItem>)}
      </TextField>
      <TextField select label="Mínima manual confirmada" value={anchors.some(a=>a.id===lowId)?lowId:''} onChange={e=>setLowId(e.target.value)} sx={{minWidth:235}}>
        <MenuItem value="">Selecione um fundo</MenuItem>{anchors.filter(a=>a.type==='BOTTOM').map(a=><MenuItem key={a.id} value={a.id}>{price(a.price)} · {time(a.extremeTimeMsc)}</MenuItem>)}
      </TextField>
      <Button onClick={applyManual} disabled={!highId||!lowId}>Aplicar Fibonacci manual</Button>
    </Stack>
    {pick && <Alert severity="info">Clique no círculo azul de um {pick==='TOP'?'topo':'fundo'} confirmado. Use a navegação para localizar a origem; os seletores também permitem escolher âncoras fora do viewport.</Alert>}
    {error && <Alert severity="warning">{error}</Alert>}
    <Typography data-testid="fibonacci-current" variant="body2">{primary
      ? `FIBO 50% — ${price(primary.fib50)} · ${mode==='MANUAL'?'Manual':'Hipótese automática'} · ${scope===5?'Micro M5':'Macro H1'} · ${primary.direction==='UP'?'Alta':'Baixa'} · máxima ${price(primary.high.price)} · mínima ${price(primary.low.price)} · decisão ${time(primary.decidedAtMsc)}`
      : mode==='MANUAL'?'Fibonacci manual aguardando duas âncoras confirmadas.':'Aguardando dupla oposta de pivôs confirmados para a hipótese automática.'}</Typography>
    <Stack direction="row">
      <FormControlLabel label="Mostrar ondas Dow micro (1+1 experimental)" control={<Checkbox checked={showMicro} onChange={(_,v)=>setShowMicro(v)}/>} />
      <FormControlLabel label="Mostrar todos os fractais micro confirmados" control={<Checkbox checked={allMicroFractals} onChange={(_,v)=>setAllMicroFractals(v)}/>} />
      <FormControlLabel label="Mostrar etiquetas HH/HL/LH/LL micro" control={<Checkbox checked={microLabels} onChange={(_,v)=>setMicroLabels(v)}/>} />
    </Stack>
    {showMicro&&!view?.microDow&&<Alert severity="info">Ondas micro indisponiveis nesta resposta do backend.</Alert>}
    {showMicro&&<Typography variant="caption">Fractais: {view?.microDow?.pivots.length??0} / Utilizados: {view?.microDow?.selectedPivotIds?.length??new Set(view?.microDow?.movements.flatMap(m=>[m.fromId,m.toId])??[]).size} / Segmentos: {view?.microDow?.movements.length??0} / Versoes historicas: {view?.microDow?.movementHistory?.length??view?.microDow?.movements.length??0}</Typography>}
    {showMicro&&view?.microDow?.opening&&<Typography variant="caption">Abertura micro: {view.microDow.opening.status} / referencia ambar separada dos pivos confirmados.</Typography>}
    {showMicro&&<details><summary>Auditoria micro: {view?.microDow?.pivots.length??0} pivos confirmados</summary><pre>{JSON.stringify(view?.microDow,null,2)}</pre></details>}
    <DowDebugPanel snapshot={view?.dow} visible={showDow} labels={dowLabels} onVisible={setShowDow} onLabels={setDowLabels}
      selectedId={selectedDow} onSelect={setSelectedDow} zone={candleDisplayTimeZone}/>
    <Stack spacing={1} data-testid="dow-fifty-panel">
      <FormControlLabel label="Mostrar 50% Dow principal" control={<Checkbox checked={showDowFifty} onChange={(_,v)=>setShowDowFifty(v)}/>} />
      <Typography variant="body2" data-testid="dow-fifty-current">{currentFifty?`DOW 50% \u2014 ${price(currentFifty.wave.level50)} / ${fiftyState[currentFifty.state]} / concluida em ${time(currentFifty.wave.recognizedAtMsc)}`:fiftyState.AGUARDANDO_ONDA}</Typography>
      <Typography variant="caption">Fonte exclusiva: ondas do Dow principal 2+2 seladas por novo pivo estrutural oposto. O segmento terminal ainda nao e onda concluida. Fibonacci antiga e manual sao independentes; a modalidade antiga continua disponivel acima para comparacao.</Typography>
      {showDowFifty&&!fifty&&<Alert severity="info">50% Dow indisponivel nesta resposta do backend. Nenhuma onda sera inferida no navegador.</Alert>}
      {showDowFifty&&<>
        <Typography variant="caption">Hipotese de rompimento: fechamento M5 alem dos 50% (experimental). Hipotese de retorno: toque e fechamento no mesmo lado (candidata nao validada). Tolerancia propria: {fifty?.approachTolerancePoints??5} pontos. Sem ordens ou destinos garantidos.</Typography>
        {currentFifty&&<Typography variant="caption">Lado observado: {currentFifty.approachSide} / Rompimento: {currentFifty.breakStatus} / Retorno: {currentFifty.rejectionStatus}. Referencias possiveis: topo {price(currentFifty.wave.high)} e fundo {price(currentFifty.wave.low)}.</Typography>}
        <TextField select label="Auditar onda 50% Dow" value={inspectedFifty?.wave.id??''} onChange={e=>setSelectedFifty(e.target.value||undefined)}>
          <MenuItem value="">Referencia vigente</MenuItem>
          {fifty?.references.map(r=><MenuItem key={r.wave.id} value={r.wave.id}>{time(r.wave.recognizedAtMsc)} / {price(r.wave.level50)} / {fiftyState[r.state]}</MenuItem>)}
        </TextField>
        {inspectedFifty&&<Alert severity="info" data-testid="dow-fifty-selected">
          Onda {inspectedFifty.wave.id} / {inspectedFifty.wave.direction} / {fiftyState[inspectedFifty.state]}.<br/>
          Inicial: {inspectedFifty.wave.initial.id}, {price(inspectedFifty.wave.initial.price)}, formacao {time(inspectedFifty.wave.initial.extremeTimeMsc)}, confirmacao {time(inspectedFifty.wave.initial.confirmedAtMsc)}.<br/>
          Final: {inspectedFifty.wave.terminal.id}, {price(inspectedFifty.wave.terminal.price)}, formacao {time(inspectedFifty.wave.terminal.extremeTimeMsc)}, confirmacao {time(inspectedFifty.wave.terminal.confirmedAtMsc)}.<br/>
          Maxima {price(inspectedFifty.wave.high)}, minima {price(inspectedFifty.wave.low)}, 50% {price(inspectedFifty.wave.level50)}. Conclusao reconhecida / linha vigente desde {time(inspectedFifty.wave.recognizedAtMsc)} / evento {inspectedFifty.wave.sourceEventId}.<br/>
          Ultima reacao: {fifty?.events.filter(e=>e.waveId===inspectedFifty.wave.id&&/TOUCH|BREAK|REJECTION|DESTINATION/.test(e.kind)).at(-1)?.kind??'Nenhuma'}. Consultar historico nao reativa referencias encerradas.
        </Alert>}
        <details><summary>Auditoria 50% Dow: {fifty?.references.length??0} ondas / {fifty?.events.length??0} eventos</summary>
          <table><thead><tr><th>Conclusao conhecida</th><th>Onda</th><th>Direcao</th><th>50%</th><th>Estado</th><th>Encerramento</th></tr></thead>
            <tbody>{fifty?.references.map(r=><tr key={r.wave.id}><td>{time(r.wave.recognizedAtMsc)}</td><td>{r.wave.id}</td><td>{r.wave.direction}</td><td>{price(r.wave.level50)}</td><td>{fiftyState[r.state]}</td><td>{r.endedAtMsc==null?'Vigente':time(r.endedAtMsc)}</td></tr>)}</tbody></table>
          {fifty?.events.map(e=><p key={e.id}>{time(e.atMsc)} / {e.kind} / {e.stateBefore} - {e.stateAfter} / {e.detail}</p>)}
          <pre>{JSON.stringify(fifty?.observations,null,2)}</pre>
        </details>
      </>}
    </Stack>
    <ThreeCandlesChart controller={controller} zone={candleDisplayTimeZone} overlay={overlay} onPick={selectAnchor}
      dowOverlay={dowOverlay} microDowOverlay={microDowOverlay} dowFiftyOverlay={dowFiftyOverlay} onPickDowFifty={setSelectedFifty} onPickDow={p=>setSelectedDow(p.extreme.id)} />
    <Typography variant="caption">Linha principal: branca no tema escuro e escura no tema claro, contínua 2,5 px; comparação: azul pontilhada.
      {' '}0% na máxima e 100% na mínima; A e B seguem a ordem dos extremos no tempo. Diagonal pontilhada e faixa azul delimitam a amplitude.
      {' '}Nenhum nível amplia o autoscale. Seleções manuais são independentes e, ao reiniciar, reaparecem quando o Replay alcança o instante em que foram feitas.</Typography>
    <details><summary>Auditoria Fibonacci: {audit.length} versões · {lane.ambiguities.length} ambiguidades ({scope===5?'M5':'H1'})</summary>
      <table><thead><tr><th>Decisão</th><th>Modo/direção</th><th>Máxima: preço, origem e confirmação</th><th>Mínima: preço, origem e confirmação</th><th>50%</th><th>Motivo</th><th>Âncoras anteriores → novas</th></tr></thead>
        <tbody>{audit.map(v=><tr key={v.id}>
          <td>{time(v.decidedAtMsc)}</td><td>{v.mode} / {v.direction}</td>
          <td>{price(v.high.price)} · {time(v.high.extremeTimeMsc)} · {time(v.high.confirmedAtMsc)}</td>
          <td>{price(v.low.price)} · {time(v.low.extremeTimeMsc)} · {time(v.low.confirmedAtMsc)}</td>
          <td>{price(v.fib50)}</td><td>{reasons[v.reason]??v.reason}</td>
          <td>{v.previous?`${v.previous.highId} / ${v.previous.lowId}`:'Inicial'} → {v.high.id} / {v.low.id}</td>
        </tr>)}</tbody>
      </table>
      {lane.ambiguities.map(a=><p key={a.id}>{time(a.atMsc)} · {a.anchor?`${a.anchor.type} ${price(a.anchor.price)} (origem ${time(a.anchor.extremeTimeMsc)}, ID ${a.anchor.id})` : ''} · {a.detail}</p>)}
      {lane.events.map(e=><p key={e.id}>{time(e.atMsc)} · {e.kind} · {e.detail} · versão {e.versionId}</p>)}
    </details>
    <details><summary>Observações geométricas dos 50%: {observations.length} · não confirmam sinais</summary>
      <p>A aproximação mede distância do intervalo do candle ao nível. Teste usa máxima/mínima; possível rejeição/rompimento usa fechamentos consecutivos. Não se presume ordem intrabar. Só se observa uma versão conhecida antes da abertura do candle.</p>
      {observations.map(o=><p key={o.id}>{time(o.atMsc)} · {o.kind} · versão {o.versionId} · {o.detail}</p>)}
    </details>
  </Stack>;
}
